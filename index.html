function countFingersForHand(landmarks, handed="Unknown"){ if(!landmarks || landmarks.length!==21) return 0; // Puntos clave const TIP = { thumb:4, index:8, middle:12, ring:16, pinky:20 }; const PIP = { index:6, middle:10, ring:14, pinky:18 }; const MCP = { thumb:2, index:5, middle:9, ring:13, pinky:17 }; let count = 0; // Detectar dedos índice, medio, anular y meñique for (const finger of ['index', 'middle', 'ring', 'pinky']) { const tip = landmarks[TIP[finger]]; const pip = landmarks[PIP[finger]]; const mcp = landmarks[MCP[finger]]; // Verificar si el dedo está extendido comparando las coordenadas Y if (tip.y < pip.y && tip.y < mcp.y) { count++; } } // Detectar pulgar const thumbTip = landmarks[TIP.thumb]; const thumbIp = landmarks[3]; const thumbMp = landmarks[MCP.thumb]; // Método mejorado para detectar el pulgar extendido if (handed === "Right") { if (thumbTip.x > thumbIp.x && thumbTip.x > thumbMp.x) { count++; } } else if (handed === "Left") { if (thumbTip.x < thumbIp.x && thumbTip.x < thumbMp.x) { count++; } } else { // Para mano desconocida, usar posición relativa if (Math.abs(thumbTip.x - thumbIp.x) > 0.05) { count++; } } return count; }
